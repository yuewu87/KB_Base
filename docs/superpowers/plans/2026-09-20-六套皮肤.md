# 六套皮肤 + 切换入口 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 Web UI 加六套可切换的配色皮肤，左栏底部一个入口点一下当场换色并记住，布局一行不动。

**Architecture:** 把 40 处写死的色值收进 CSS 变量层 → 每套皮肤就是一组同名变量的重定义，挂在 `<html data-skin>` 上。六套的取值由一个纯 Python 模块 `src/kb/web/skins.py` 从 6 个锚点推导出来（26 个推导值全是固定比例，无手调），`scripts/derive_skins.py` 把它写进 `style.css`，测试断言两边一致——**CSS 里那 192 个数字不许手改**。当前皮肤存 `KB_SKIN`，走现有的 `/settings` 落盘链路，不用重启。

**Tech Stack:** FastAPI + Jinja2 + 单文件 `style.css`（无构建步骤、无前端框架、无外链字体）；测试用 pytest + `fastapi.testclient`；conda 环境 `kn_base`。

**设计来源：** `docs/superpowers/specs/2026-09-20-六套皮肤-design.md`（已过目，三条取舍按推荐执行）

```python
"""六套皮肤的取值、CSS 落地、以及「别再写死色值」的防回潮。"""

import re
from pathlib import Path

STYLE_CSS = Path(__file__).resolve().parents[2] / "src" / "kb" / "web" / "static" / "style.css"

# 允许出现颜色字面量的地方：`:root` 里那一段变量定义。
# 这个列表随皮肤块一起由 scripts/derive_skins.py 生成，见 Task 3。
HEX = re.compile(r"#[0-9a-fA-F]{3,8}\b")


def test_no_hardcoded_hex_outside_variable_blocks():
    """变量定义之外不许出现十六进制色值。

    **这条是防回潮的**：以后随手写一个 `color: #fff`，换皮就会在那处漏出
    上一套皮肤的颜色——而这种事在截图里不一定看得出来。
    """
    text = STYLE_CSS.read_text(encoding="utf-8")
    # 先把变量定义的行去掉（`--x: #abc;` 这种），再看还剩多少
    leftovers = []
    for n, line in enumerate(text.splitlines(), 1):
        if line.strip().startswith("--"):
            continue
        if HEX.search(line):
            leftovers.append(f"{n}: {line.strip()}")
    assert leftovers == [], "变量定义之外还有写死的色值：\n" + "\n".join(leftovers)
```

```python
"""六套皮肤的取值：从每套 6 个锚点推出其余 26 个变量。

## 为什么是「推导」而不是「手挑」

31 个变量 × 6 套 = 186 个十六进制数字。手挑的话，调一次「② 的紫再深一点」
就要跟着改二十个配套值，改漏一个就是一处对不上——而且看不出来。
所以除了用户定的 6 个锚点，其余**全部按固定比例算**，规则写在下面，
一个「看着调」都没有。

`scripts/derive_skins.py` 把 `render_blocks()` 的结果写进 `style.css`，
测试断言两边逐字相等。**CSS 里那六块不要手改。**

## ⑤ 不参与推导

用户要求「原样保留」，所以 `ARCHIVE` 把今天线上生效的 32 个值全部硬编码。
推导规则以后怎么改都不许波及它——`test_archive_is_verbatim_todays_values` 守着。

## 对比度

强调色有两个用途，对对比度的要求不一样：**当填充**（按钮底）只要上面的字看得清；
**当文字**（链接、选中项）它自己要 4.5:1。② 的紫、⑥ 的蓝当文字只有 2.85 / 2.60:1，
所以多出一档 `--accent-ink`（朝可读方向推到 4.5），**锚点不动**。
"""

from __future__ import annotations

from dataclasses import dataclass

# 深色皮要认，浏览器才会把滚动条、<select> 下拉、readonly 输入框跟着画深
_LIGHT = "light"
_DARK = "dark"


@dataclass(frozen=True)
class Skin:
    id: str
    name: str          # 面板上显示的名字
    paper: str
    surface: str
    ink: str
    line: str
    accent: str
    accent_soft: str
    second: str | None  # 撞色里的第二个颜色：只当「巡检金」用，不当第二个强调色
    tone: str           # _LIGHT / _DARK

    @property
    def dark(self) -> bool:
        return self.tone == _DARK


# 六个锚点由用户定稿（`.superpowers/brainstorm/.../palettes.html`），不许改
SKINS: tuple[Skin, ...] = (
    Skin("archive", "现有蓝", "#fbfcfe", "#ffffff", "#1a2233", "#e4e9f0",
         "#2f5d8a", "#eef4fb", None, _LIGHT),
    Skin("dark-pink", "炭黑甜酷粉", "#1a1a1d", "#242429", "#f0f0f3", "#35353d",
         "#e6397c", "#3b1b2b", None, _DARK),
    Skin("aurora", "极光紫蜜柚黄", "#fbfaff", "#ffffff", "#2a2440", "#e7e3f5",
         "#9f82fd", "#efeaff", "#fbea03", _LIGHT),
    Skin("garnet", "石榴红雾粉桃", "#fdf8f8", "#ffffff", "#33191e", "#f1dddf",
         "#e72d48", "#fbe4e7", "#f1dddf", _LIGHT),
    Skin("neon", "极光紫荧光绿", "#241c3a", "#2e2549", "#efeafb", "#3e3462",
         "#bcfe1a", "#3a3457", "#9f82fd", _DARK),
    Skin("mono", "黑皮", "#16171b", "#1f2025", "#e9ebef", "#2e3037",
         "#2f5d8a", "#23303f", None, _DARK),
)

SKIN_IDS: tuple[str, ...] = tuple(s.id for s in SKINS)
DEFAULT_SKIN = "archive"
_BY_ID = {s.id: s for s in SKINS}

# 展示顺序：用户定的 ①-⑥ 顺序，不是上面的定义顺序
_DISPLAY_ORDER = ("dark-pink", "aurora", "garnet", "neon", "archive", "mono")


# ----------------------------------------------------------------- 颜色小工具

def _rgb(color: str) -> tuple[int, int, int]:
    h = color.lstrip("#")
    return int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16)


def _hex(rgb) -> str:
    return "#%02x%02x%02x" % tuple(max(0, min(255, round(c))) for c in rgb)


def mix(a: str, b: str, t: float) -> str:
    """`a` 朝 `b` 线性插值 `t`（在 gamma 编码的 sRGB 上，与 CSS `color-mix` 一致）。"""
    ra, ga, ba = _rgb(a)
    rb, gb, bb = _rgb(b)
    return _hex((ra + (rb - ra) * t, ga + (gb - ga) * t, ba + (bb - ba) * t))


def _luminance(color: str) -> float:
    def channel(v: int) -> float:
        v /= 255
        return v / 12.92 if v <= 0.03928 else ((v + 0.055) / 1.055) ** 2.4

    r, g, b = _rgb(color)
    return 0.2126 * channel(r) + 0.7152 * channel(g) + 0.0722 * channel(b)


def contrast(a: str, b: str) -> float:
    """WCAG 对比度。"""
    la, lb = _luminance(a), _luminance(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)


def _rgba(color: str, alpha: float) -> str:
    r, g, b = _rgb(color)
    return f"rgba({r}, {g}, {b}, {alpha})"


def _pick_ink(bg: str) -> str:
    """压在 `bg` 上的字用白还是用黑。

    **白字达标（≥4.5:1）就用白，否则才用黑**——不是简单地「谁高选谁」。
    ③ 白 4.32 / 黑 4.58，差 0.26，纯粹比大小会翻成黑字，但「白字压红底」
    是更强的视觉习惯，0.26 的差别也谈不上更好读。定成阈值就没这个摇摆。
    """
    return "#ffffff" if contrast("#ffffff", bg) >= 4.5 else "#0a0a0c"


def _to_contrast(accent: str, paper: str, target: float, toward: str) -> str:
    """把 `accent` 朝 `toward` 推，直到对 `paper` 的对比度达标。"""
    if contrast(accent, paper) >= target:
        return accent
    for step in range(1, 256):
        candidate = mix(accent, toward, step / 256)
        if contrast(candidate, paper) >= target:
            return candidate
    return toward

def _derive(skin: Skin) -> dict[str, str]:
    paper, surface = skin.paper, skin.surface
    ink, line = skin.ink, skin.line
    accent, accent_soft = skin.accent, skin.accent_soft
    dark = skin.dark

    values = {
        "--paper": paper,
        "--surface": surface,
        "--ink": ink,
        "--line": line,
        "--accent": accent,
        "--accent-soft": accent_soft,

        "--faint": mix(ink, paper, 0.50),
        "--ink-soft": mix(ink, paper, 0.16),

        "--line-soft": mix(line, surface, 0.45),

                               .28 if dark else .12),
        # 强调色当「文字」用：推到 4.5:1。浅色皮往墨推、深色皮往白推。
        "--accent-ink": _to_contrast(accent, paper, 4.5,
                                     ink if not dark else "#ffffff"),

def render_vars(skin_id: str, indent: str = "  ") -> str:
    """把 32 个变量渲染成 `--x: y;` 的行，用于 `:root` 里面。"""
    values = derive(skin_id)
    return "\n".join(f"{indent}{key}: {values[key]};" for key in ORDER)


def render_block(skin_id: str) -> str:

    skin = _BY_ID[skin_id]
    tone = "dark" if skin.dark else "light"
    return (
        f'[data-skin="{skin_id}"] {{   /* {skin.name} */\n'
        f"  color-scheme: {tone};\n"
        f"{render_vars(skin_id)}\n"
        f"}}\n"
    )


def options() -> tuple[dict[str, str], ...]:

    return tuple(
        {"id": sid, "name": _BY_ID[sid].name, "dot": derive(sid)["--accent"]}
```

```python
"""把 `kb.web.skins` 渲染的皮肤块写进 `style.css`。

用法：

    "D:/Conda_base/envs/kn_base/python.exe" scripts/derive_skins.py          # 只打印差异
    "D:/Conda_base/envs/kn_base/python.exe" scripts/derive_skins.py --write  # 真的写

**`--write` 是幂等的**：连跑两次第二次什么都不改。

改配色请改 `src/kb/web/skins.py` 里的规则或锚点，**不要手改 style.css 里那几块**
——`tests/web/test_skins.py` 会当场发现两边对不上。
"""

from __future__ import annotations

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1] / "src"))

from kb.web import skins  # noqa: E402  （要先加好 sys.path）

STYLE_CSS = Path(__file__).resolve().parents[1] / "src" / "kb" / "web" / "static" / "style.css"

BEGIN = "/* ===== 皮肤：由 scripts/derive_skins.py 生成，不要手改 ===== */"
END = "/* ===== 皮肤块结束 ===== */"

MARKED = (
    BEGIN
    + "\n\n"
    + "\n".join(skins.render_block(sid) for sid in skins.SKIN_IDS if sid != "archive")
    + "\n"
    + END
    + "\n"
)


def current_region(text: str) -> str:
    """取出 style.css 里那段生成区（含标记）。没有就返回空串。"""
    if BEGIN not in text or END not in text:
        return ""
    start = text.index(BEGIN)
    stop = text.index(END) + len(END) + 1
    return text[start:stop]


def apply(text: str) -> str:
    """把生成区换上最新内容。没有生成区就追加到文件末尾。"""
    region = current_region(text)
    if region:
        return text.replace(region, MARKED)
    return text.rstrip("\n") + "\n\n" + MARKED


def main() -> int:
    write = "--write" in sys.argv
    text = STYLE_CSS.read_text(encoding="utf-8")
    updated = apply(text)

    if updated == text:
        print("style.css 已经是最新的")
        return 0

    if not write:
        print("style.css 与 skins.py 不一致（加 --write 才会真的写）")
        return 1

    STYLE_CSS.write_text(updated, encoding="utf-8", newline="\n")
    print(f"已更新 {STYLE_CSS}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

- [ ] **Step 4: 加 JS**

`base.html` 第二个 `<script>` 里（`quitService` 之后、`</script>` 之前）追加：

```js
    // ---- 外观面板 ----
    //
    // 面板不在侧栏里面（侧栏 `overflow-y: auto` 会把横向伸出去的浮层裁掉），
    // 所以在 body 下用 `position: fixed`，位置按按钮的 rect 现算。
    //
    // **先改 DOM 再发请求**：`data-skin` 一改页面当场变色，不用等网络；
    // 请求失败只影响「下次打开是不是这套」，界面不会卡一下。
    function _skinParts() {
      return {
        btn: document.getElementById('skin-btn'),
        menu: document.getElementById('skin-menu'),
      };
    }

    function closeSkins() {
      const { btn, menu } = _skinParts();
      if (!btn || !menu) return;
      menu.hidden = true;
      btn.setAttribute('aria-expanded', 'false');
    }

    function toggleSkins() {
      const { btn, menu } = _skinParts();
      if (!btn || !menu) return;
      if (!menu.hidden) { closeSkins(); return; }
      // 贴着按钮的右边、底边对齐。折叠成 60px 时按钮还在，所以照样能开。
      const r = btn.getBoundingClientRect();
      menu.style.left = (r.right + 8) + 'px';
      menu.style.bottom = Math.max(8, window.innerHeight - r.bottom) + 'px';
      menu.hidden = false;
      btn.setAttribute('aria-expanded', 'true');
      const on = menu.querySelector('.skin-item.on');
      if (on) on.focus();
    }

    function pickSkin(id) {
      document.documentElement.dataset.skin = id;   // 当场变色

      const menu = document.getElementById('skin-menu');
      menu.querySelectorAll('.skin-item').forEach(function (el) {
        const mine = el.dataset.skinId === id;
        el.classList.toggle('on', mine);
        el.setAttribute('aria-checked', mine ? 'true' : 'false');
      });
      closeSkins();

      fetch('/settings', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({values: {KB_SKIN: id}}),
      }).then(function (r) {
        // **`fetch` 只在网络层失败时 reject**，4xx/5xx 都是「成功」的响应
        if (!r.ok) throw new Error('HTTP ' + r.status);
      }).catch(function (e) {
        // 界面已经变了，所以这里只影响「下次打开是不是这套」——
        // 但**不能一声不吭**，否则用户以为存上了
        const msg = document.getElementById('skin-msg');
        msg.textContent = '皮肤没存上：' + e;
        msg.hidden = false;
        menu.hidden = false;
      });
    }

    // 点面板外面、按 Esc 都关掉。`window` 的 resize 也关——
    // 位置是算出来的，窗口一变就对不准了，不如关掉重开。
    document.addEventListener('click', function (e) {
      const { btn, menu } = _skinParts();
      if (!menu || menu.hidden) return;
      if (btn.contains(e.target) || menu.contains(e.target)) return;
      closeSkins();
    });
    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape') closeSkins();
    });
    window.addEventListener('resize', closeSkins);
```
